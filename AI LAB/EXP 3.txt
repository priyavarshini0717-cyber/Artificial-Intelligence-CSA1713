from collections import deque
def water_jug():
    queue = deque([((0, 0), [])])
    visited = set()
    while queue:
        (a, b), path = queue.popleft()
        if (a, b) in visited:
            continue
        visited.add((a, b))
        path = path + [(a, b)]
        if a == 2:
            print("Solution:")
            for state in path:
                print(state)
            return
        states = [
            (4, b),  # Fill 4L jug
            (a, 3),  # Fill 3L jug
            (0, b),  # Empty 4L jug
            (a, 0),  # Empty 3L jug
            (a - min(a, 3 - b), b + min(a, 3 - b)),
            (a + min(b, 4 - a), b - min(b, 4 - a))
        ]
        for state in states:
            if state not in visited:
                queue.append((state, path))
water_jug()

from collections import deque
def valid(m, c):
    if m < 0 or c < 0 or m > 3 or c > 3:
        return False
    if m > 0 and m < c:
        return False
    if 3 - m > 0 and 3 - m < 3 - c:
        return False
    return True
def solve():
    start = (3, 3, 0)
    goal = (0, 0, 1)
    queue = deque([(start, [])])
    visited = set()
    while queue:
        state, path = queue.popleft()
        if state in visited:
            continue
        visited.add(state)
        path = path + [state]
        if state == goal:
            for s in path:
                print(s)
            return
        m, c, boat = state
        moves = [(1, 0), (2, 0), (0, 1), (0, 2), (1, 1)]
        for dm, dc in moves:
            if boat == 0:
                new_state = (m - dm, c - dc, 1)
            else:
                new_state = (m + dm, c + dc, 0)
            if valid(new_state[0], new_state[1]):
                queue.append((new_state, path))
solve()
